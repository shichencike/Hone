#5. 控制流

条件表达式必须显式返回 bool，禁止隐式转换（if (1) 视为非法）。

x = 7;                  // 先定义再判断
if (x > 5) {
    print("大");
} else if (x > 0) {
    print("小");
} else {
    print("非正");
}

i = 0;
while (i < 10) {
    i = i + 1;
}

// for-in：遍历列表（单变量）或字典（键、值双变量）
nums = [1, 2, 3];
for x in nums {
    print(x * x);      // 1 4 9
}

user = {"name": "hone", "ver": 1};
for k, v in user {
    print(k + "= " + to_str(v));
}

// do-while：先执行循环体再判断条件（至少执行一次）
j = 0;
do {
    j = j + 1;
} while (j < 3);

// C 风格 for：init; cond; step 三段均可省略（for (;;) 无限循环，配合 break 退出）
for (i = 0; i < 5; i = i + 1) {
    print(i);          // 0 1 2 3 4
}

// continue：跳过本次循环剩余语句，直接进入下一次迭代
for (i = 1; i <= 5; i = i + 1) {
    if (i % 2 == 0) { continue; }
    print(i);          // 1 3 5（只打印奇数）
}

标签与 goto（H105）

Hone 提供编译期安全的 goto：用 名字: 定义标签（单独占一行、行尾紧跟冒号），用 goto 名字; 跳转。常用于手工搭循环、一次性跳出多层嵌套、跳过一段代码。

i = 0;
again:
print(i);
i = i + 1;
if (i < 3) { goto again; }   // 回跳：依次打印 0 1 2

// 向外层标签跳转：一次性跳出嵌套循环（等价于带标签 break）
for b in [2, 4] {
    for a in [1, 2, 3] {
        if (a == 3) { goto found; }
    }
}
found:
print(b);   // 2（a==3 时直接跳出两层循环）

安全规则（全部编译期检查，运行期绝不跳到未定义位置，报错 error[H105]）：

  • 标签只能在同一函数内使用，且同一函数内名字唯一（重复定义报错）；
  • 只能跳到「从跳转点向外层可见」的标签：标签必须位于跳转点所在语句块或其外层语句块中，因此无法跳进 if / 循环等内层语句块（也就不存在「跳入未初始化作用域」的问题）；
  • 禁止向前跳过变量绑定：若标签在跳转点之后，且两者之间在同一语句块内存在 x = ... / int x = ... / 解构赋值等绑定语句，则报错（否则跳转后会读到未初始化变量）；
  • 标签不跨函数：lambda、嵌套 fn 各自独立，外层标签在内层不可见（反之亦然）；
  • 标签写法 loop_top: 与类型注解 x : int = 10; 不冲突：冒号后同一行还有内容即按类型注解解析；
  • AOT 原生编译（build --exe -c）暂不支持 goto，报 error[H999]（尚未实现）并提示改用 hone run / hone run --vm；解释器与字节码 VM 均完整支持。

排查标签 / 跳转报错：hone explain H105。

宏 macro（H106）

宏是解析期的代码生成工具：在解析之后、类型检查之前，由预处理阶段（src/preproc.rs）做纯 AST 替换。因此解释器 / 字节码 VM / AOT 三个后端看到的是展开后的同一份代码，运行期不存在宏实体，行为完全一致。

两种形式，调用写法均与函数一致（NAME(实参)）：

// 表达式宏：用于表达式位置
macro SQ(x) => x * x;
macro CLAMP(v, lo, hi) => (v < lo) ? lo : ((v > hi) ? hi : v);
print(SQ(3));                   // 9
print(CLAMP(12, 0, 10));         // 10

// 语句宏：展开为独立作用域代码块
macro SAY(msg) { print(msg); }
macro INC(x) { x = x + 1; }      // 形参作赋值目标：实参须是裸变量名
SAY("hi");
n = 0; INC(n); print(n);           // 1

安全约定（编译期检查，越界用法报 error[H106]）：

  • 宏必须定义在程序顶层（函数 / 代码块 / 类 / 循环内定义宏一律报错）；
  • 先定义后使用：按源码顺序展开；若调用的名字对应更靠后的宏定义，直接报 H106 并指出定义行号（提示把定义上移）；
  • 宏体在定义点展开：只能引用比它更早定义的宏 ⇒ 引用关系必然无环，不存在无限展开，也不需要展开深度上限；
  • 纯 AST 替换（非文本替换）：无运算符优先级陷阱、无 token 拼接问题；
  • 宏名不能与函数 / async fn / 结构体 / 类 / 枚举重名，宏之间也不能重名；
  • 实参个数必须与形参个数一致；
  • 表达式宏体内禁用 ++ / -- 与 lambda（会改变实参的求值作用域）；形参不能作为被调用的函数名（宏是语法替换，不支持高阶）；
  • 语句宏展开为独立作用域块，体内声明的变量不会泄漏到调用处；体内禁止 return / goto / 标签 / 嵌套 macro；break / continue 仅限宏体自身的循环；
  • 形参可用作简单赋值目标（x = ... / x += ... / int x = ...），此时对应实参必须是裸变量名；不能用作循环变量、catch 变量、解构目标或类型名。

与函数的分工：需要返回值、递归、一等值（传参 / 存变量）时用 fn；只想去掉一层样板写法时用宏。排查宏报错：hone explain H106。

#6. 函数

fn add(a : int, b : int) -> int {
    return a + b;
}

// 参数类型可由调用上下文推导（推荐复杂场景显式声明）
fn greet(name) {
    print("你好, " + name);
}
  • 函数返回类型由 return 推导，也可用 fn f() -> type 声明
  • 无返回值时调用作为表达式得到 null 占位
  • 所有函数扁平化存在于全局符号表（不支持 :: 路径）

泛型函数

在函数名后、参数列表前用 [T, U] 声明类型参数，参数与返回类型注解可引用 （x: T、-> T）。调用时按实参自动推导，同一泛型函数 可以不同类型多次调用、互不锁定；编译期擦除——运行期与普通函数零差异，无运行时开销、不产生重复代码。

fn identity[T](x: T) -> T {
    return x;
}
print(identity(42));          // 42（T=int）
print(identity("hello"));     // hello（T=str）

fn swap[A, B](a: A, b: B) -> B {
    return b;
}
print(swap(1, "two"));        // two（A=int, B=str）

fn pick[T](a: T, b: T, want_first: bool) -> T {
    if (want_first) {
        return a;
    }
    return b;
}
print(pick("a", "b", true)); // a（T=str）
print(pick(10, 20, false));  // 20（T=int）

约束：函数名全局唯一（泛型与普通同名仍报 already defined）；未声明的类型参数报 H002； 重复类型参数报 H005；同一类型参数处调用点必须传同型（否则 H001）。 类方法同样支持泛型（class 中 fn 方法[T](...)）。 完整示例见 examples/generic_test.hn。

默认参数

fn f(a, b = 10)：调用时省略的尾部实参取默认值。 必选参数（无默认值）必须位于默认参数之前（否则报错）；默认表达式在调用时求值， 可引用其前面的参数（如 fn f(a, b = a * 2)）。参数个数检查按 「必选个数 ~ 总个数」区间；lambda 参数同样支持默认值；hone build --dll 暂不支持。

fn greet(name, greeting = "Hello") {
    return greeting + ", " + name + "!";
}
print(greet("Hone"));           // Hello, Hone!
print(greet("Hone", "Hi"));     // Hi, Hone!

fn repeat_char(ch, times = 3, sep = "-") {
    s = "";
    for (i = 0; i < times; i = i + 1) {
        s = s + ch;
        if (i < times - 1) { s = s + sep; }
    }
    return s;
}
print(repeat_char("*"));        // *-*-*
print(repeat_char("*", 2));     // *-*

匿名函数（lambda）

fn(x) { ... } 是一等值：可赋值给变量、作为参数传递（高阶函数）、 作为返回值返回（闭包工厂）。创建时按值捕获当前作用域可见的变量（闭包）， 调用时经变量名动态解析；无函数名、无泛型、无 -> 返回注解，返回类型动态。 break / continue / return 只能出现在 lambda 自身函数体内。

double = fn(x) {
    return x * 2;
};
print(double(21));     // 42

base = 10;
adder = fn(y) {
    return base + y;   // 捕获外围变量 base（按值）
};
print(adder(5));       // 15

fn apply(f, v) {
    return f(v);       // f 为 lambda 参数，动态调用
}
print(apply(fn(n: int) { return n * n; }, 6));   // 36

fn make_adder(offset) {
    return fn(z) { return z + offset; };   // 返回闭包
}
add5 = make_adder(5);
print(add5(10));       // 15

#7. 集合与字符串

列表 [a, b, c] 与字典 {"k": v} 为动态集合类型：元素类型不锁定、可混合， 变量本身仍遵循类型锁定规则（不能把列表赋给已锁定的标量变量）。

nums = [1, "a", true];          // 混合元素列表
user = {"name": "hone", "ver": 1};   // 字符串键 → 任意值

print(len(nums));               // 3（元素个数）
print(to_str(user));            // {name: hone, ver: 1}

l = append(nums, "x");          // [1, a, true, x]（返回新列表）
print(contains(l, "a"));       // true
print(index_of(l, "x"));       // 3（找不到返回 -1）

print(has_key(user, "name"));  // true
print(to_str(keys(user)));      // [name, ver]
print(to_str(values(user)));    // [hone, 1]

f-string 字符串插值：f"..." 内 {expr} 嵌入任意表达式， {{ / }} 转义字面大括号：

name = "hone";
print(f"你好, {name}! sum = {1 + 2}");   // 你好, hone! sum = 3
print(f"{{{1 + 1}}}");                 // 字面大括号 + 表达式

三引号原始字符串："""...""" 多行字符串，内容不做转义处理 （\n、\t 等原样保留），跨行书写、保留换行与缩进；与普通字符串同值 （支持 len、拼接、比较等）：

text = """line1
line2
    indented""";
print(text);       // line1 换行 line2 换行 "    indented"
print(len(text));  // 24

类型判断：is_int is_float is_str is_bool is_list is_dict is_null，判断运行期值类型，常用于分支处理混合集合。

#7.5 结构体 · 模式匹配 · 管道

struct 结构体：声明确定的数据形态（字段名与类型固定），实例用 dict 表示、字段访问 p.字段； 构造时按字段顺序传参，检查阶段校验字段个数与类型（H001/H011）：

struct Point { x: int, y: float };
p = Point(3, 2.5);
print(p.x);   // 3
print(p.y);   // 2.5

match 模式匹配：多分支表达式的简洁写法，模式支持字面量与 _ 通配符（放最后、只能一次）， 返回匹配分支的值，全部不匹配时运行时报错（建议补 _ 兜底）：

s = match 2 {
    1 => "one",
    2 => "two",
    _ => "other",
};
print(s);   // two

管道操作符 |>：数据流向更直观。x |> f 等价 f(x)， x |> f(a, b) 等价 f(x, a, b)，可链式（解析期转换为普通调用）：

print("hello world" |> len);       // 11
print([1, 2, 3] |> len |> to_str); // "3"

#7.6 class 类（成员函数不进全局符号表）

class 类：将一组相关函数组织成命名空间，通过 类名.方法(...) 调用 （如 Math.double(21)）。关键特性：成员函数不进入全局符号表—— 不能用裸名调用（double(21) 报 H002），不污染全局命名空间， 同名全局函数可与类方法共存、互不影响；类内方法互相调用也要用限定名 （return Math.fib(n - 1) + Math.fib(n - 2);）。

class Math {
    fn double(x) {
        return x * 2;
    }
    fn fib(n) {
        if (n <= 1) {
            return n;
        }
        return Math.fib(n - 1) + Math.fib(n - 2);
    }
    fn greet(name) -> str {
        return "Hello, " + name;
    }
}

print(Math.double(21));   // 42
print(Math.fib(10));      // 55
print(Math.greet("hone")); // Hello, hone

fn double(x) {            // 全局同名函数与类方法共存
    return "global: " + to_str(x);
}
print(double(21));        // global: 21
print(Math.double(21));   // 42（类方法不受影响）

类方法支持参数、返回值（可写 -> type 注解）、递归、return，与普通函数能力一致； 类名与 struct 名不能重复（报 H005）；结尾分号可选（class 是块结构，与 struct 不同）。 完整示例见 examples/class_test.hn。

#7.7 多返回值 · 解构赋值 · 推导式 · 可选链

多返回值：return a, b, ...; 一个函数可一次返回多个值， 运行时打包为列表，由解构赋值接收；不解构时（x = f();）拿到打包后的列表。 仍支持单值 return 与空 return（返回 null）；hone build --dll 暂不支持：

fn divmod(a, b) {
    return a / b, a % b;
}
q, r = divmod(10, 3);
print(q);   // 3
print(r);   // 1

fn minmax(a, b) {
    if (a < b) { return a, b; }
    return b, a;
}
lo, hi = minmax(7, 2);
print(lo);   // 2
print(hi);   // 7

packed = divmod(8, 3);
print(packed);   // [2, 2]（不解构时拿到打包列表）

解构赋值：列表解构 a, b = [1, 2] 按位置依次绑定变量 （元素不足报错、多余元素忽略，典型用途 a, b = [b, a]; 交换）； 字典解构 {a, b} = dict 按键取出同名变量，改名形式 {a: x, b: y} = dict（等价 x = dict["a"]），键缺失报错：

a, b = [1, 2];          // a=1, b=2
first, second = [9, 8, 7];  // 多余元素忽略：first=9, second=8

user = {"name": "hone", "age": 7};
{name, age} = user;           // name="hone", age=7
{name: n, age: a} = user;     // 改名：n="hone", a=7

q, r = divmod(17, 5);     // 与多返回值配合：q=3, r=2

列表/字典推导式：[表达式 for 变量 in 集合 (if 条件)] 与 {键: 值 for 变量 in 集合 (if 条件)}，一行生成集合。 集合为列表时单变量绑定元素，为字典时双变量（k、v 分别为键、值）； 字典推导式的键必须求值为 str（动态键用 to_str 转换）：

nums = [1, 2, 3, 4, 5];
dbl = [x * 2 for x in nums];              // [2, 4, 6, 8, 10]
evens = [x for x in nums if x % 2 == 0];  // [2, 4]

grades = {"alice": 90, "bob": 85};
passed = [k for k, v in grades if v >= 90];   // ["alice"]
rev = {to_str(v): k for k, v in grades};      // {"90": "alice", "85": "bob"}

可选链 ?.：a?.b 在 a 为 null 时 短路返回 null，否则等价普通字段访问；可链式 a?.b?.c，可与空值合并配合 a?.b ?? "默认值"。混合链 a?.b.c 中 ?. 只短路其后一个字段，继续的普通字段访问在 null 上仍报错（与 JS 一致）。 obj 非 null 但字段不存在时同样报错；Hone 无 null 字面量，null 来自 void 函数调用等占位值； hone build --dll 暂不支持：

fn get_user() {
    return {"name": "hone", "age": 7};
}
print(get_user()?.name);   // hone

fn void_fn() {
}
v = void_fn();
print(v?.name);             // null（短路）
print(v?.a?.b);             // null（链式短路）
print(v?.name ?? "匿名");  // 匿名（与空值合并配合）

#7.8 枚举（enum）与 match 变体模式

enum：enum 名称 { 变体, 变体(类型, ...), ... }; 简单变体无载荷、带载荷变体携带多个类型化字段（int/float/bool/str）。变体访问 Color.Red（无载荷）或构造 Shape.Circle(1.5) （带载荷，按载荷顺序传参，检查阶段校验个数与类型）。==/!= 比较类型 + 变体 + 载荷（不同枚举互不相等）。枚举名全局唯一（与 struct/class 重名报错）， 变体名在枚举内唯一。值显示为 Color.Red / Shape.Circle(1.5)。

match 变体模式：Color.Red（无载荷）或 Shape.Circle(r)（载荷绑定到分支体变量，_ 忽略某项）， 与字面量 / _ 通配符模式共存；绑定变量仅本分支内可见：

enum Shape { Circle(float), Rect(float, float) };
area = match Shape.Circle(2.0) {
    Shape.Circle(r) => 3.14159 * r * r,
    Shape.Rect(w, h) => w * h,
    _ => 0,
};
print(area);   // 12.56636…

全链路支持解析 / 静态检查 / 解释器 / AOT 原生编译 / fmt / LSP；示例 examples/enum_demo.hn。

#7.9 运算符重载（__op 特殊函数）

顶层定义特殊函数 fn __op(参数) { ... }，仅在内建运算不支持的 操作数组合时被调用；内置类型（int/float/str/list/dict/enum 等）的固有行为完全不受影响。 覆盖二元 + - * / % == != < <= > >= （__add/__sub/__mul/__div/__mod/__eq/__ne/__lt/__le/__gt/__ge）、 一元 -x/!x （__neg/__not）、内置 len(x)/ x[i]（__len/__index）。 重载函数与普通函数一样遵循返回类型锁定；定义了重载后涉及动态值的运算静态检查放宽 （交给运行时判定）。AOT 原生编译支持二元/一元重载（生成「内建组合 → 内建运算，否则 → __op 调用」分派链）；示例 examples/overload_demo.hn：

fn __add(a, b) {
    if (type_of(a) == "dict" && type_of(b) == "dict") {
        return mk_complex(c_re(a) + c_re(b), c_im(a) + c_im(b));
    }
    return {"re": 0, "im": 0};
}
zs = z1 + z2;        // dict + dict 内建不支持 → __add
print(to_str(zs));  // {re: 4, im: 6}

#7.10 异步函数（async fn + await）

async fn 名称(参数) { ... } 定义异步函数（与 fn 同构）。 调用时在后台线程执行并立即返回 future（不阻塞调用方）；await 表达式 阻塞等待 future 完成并返回其结果。错误传播：async 函数体内 throw / 运行时错误存入 future， await 时原样抛出（可 try/catch 捕获）。与 go 互补： go 发后不理、async/await 结构化并发（多个 async 并行启动、依次 await，总耗时约等于最慢者）。 AOT 与 hone build --dll 对 async/await 报「暂不支持」； 示例 examples/async_demo.hn：

async fn compute(n) {
    return n * 2;
}
f = compute(21);      // 立即返回 future（不阻塞）
print(await f);    // 42（阻塞等待结果）

输入默认值（EOF 降级）

input / read_int / read_float 在 EOF（管道关闭 / Ctrl+Z / Ctrl+D）时抛 error[H306]， read_int / read_float 格式非法时抛 error[H006] / error[H007]。 它们不返回 null，因此不能直接用 ?? 空值合并兜底； 给输入设默认值的标准做法是用 try-catch 包装：

fn ask(prompt, fallback) {
    try {
        return input(prompt);
    } catch e {
        return fallback;   // H306 EOF → 返回默认值
    }
}
name = ask("姓名: ", "匿名");
print("你好, " + name + "!");

// 数字输入同理：一个 catch 同时兜住格式错误 H006/H007 与 EOF H306
fn ask_int(prompt, fallback) {
    try {
        return read_int(prompt);
    } catch e {
        return fallback;   // H006 格式错 / H306 EOF → 返回默认值
    }
}
age = ask_int("年龄: ", 0);
print("明年你 " + to_str(age + 1) + " 岁");

注意：?? 空值合并只对 null 值生效（如 void 函数调用占位值）； input 在 EOF 是抛错误而不是返回 null，必须用 try-catch 捕获后降级。

模块清单 hone.json（类似 package.json / Cargo.toml）

项目根目录放一个 hone.json，即可集中声明全部模块依赖（含元信息）， 换机器 / 分享项目时用一条命令拉齐所有模块：

{
"name": "myapp",      // 项目名（新建清单时自动取目录名）
"version": "0.1.0",   // 项目版本
"modules": {        // 模块依赖：模块名 → 源码 URL（或本地路径）
    "math_mod": "https://example.com/math_mod.hn",
    "gui":     "./hone_lib/gui.hn"
}
}

用法：

  • hone get（不带参数）：读取当前目录 hone.json， 批量下载全部模块到 ~/.hone/cache/（已缓存则跳过）
  • hone get <module> <url>：下载单个模块， 并把 (module, url) 追加写入 hone.json（已存在则更新其 URL；清单不存在则自动新建）
  • hone get <script.hn>：扫描脚本内所有 import 声明批量预下载

要点：模块 URL 非 http/https 开头按本地路径处理，直接读取源码（不联网）， 便于同仓库共享 hone_lib/ 等模块； 清单缺失报 error[H404]，JSON 语法错误 / modules 为空报 error[H005]。

typed FFI（v0.4+）：签名块内 fn 声明仅描述 C 函数 ABI，不定义 Hone 函数体。 支持类型映射：int → int64_t、float → double、bool → C bool、 str → const char*、ptr → void*；返回类型额外支持 void； 参数最多 8 个，回调 fn(...) 与可变参数 ... 暂不支持。 调用在检查阶段校验参数个数与类型（H001/H005），返回类型参与类型推导。 未声明签名的库函数保持旧 int64 通道调用（向后兼容）。

头文件自动绑定（v0.4+）：load ... from "header.h" 用受限解析器 （纯 Rust，无 libclang 依赖）从 C 头文件提取函数原型自动生成签名：跳过注释/预处理/struct 定义/extern "C" 块， typedef 简单展开，属性宏跳过；类型映射 int/size_t→int、float/double→float、bool→bool、char*→str、 其余指针→ptr、void→void。回调/变参/数组/结构体按值/long double 标记 unsupported，调用时直接报错而非 ABI 崩溃。 hone bind <header.h> 离线生成签名块输出到 stdout，unsupported 原型列为注释。

示例：examples/ffi_demo.hn（typed FFI 全类型）、examples/ffi_header.hn（头文件绑定）。

AOT 原生编译与 C 后端架构

hone build --exe -c 把脚本翻译为 C 中间文件， 再交给系统 C 编译器产出原生可执行文件。三个执行后端（AST 解释器、寄存器式字节码 VM、AOT） 语义一致，差别只在性能与部署形态。

实测性能

相对解释器有接近一个数量级的稳定加速 （MinGW-w64 gcc 16.2.0，预热后取多轮最快值），产物体积约 58.6 KB、零运行时依赖：

基准AOT解释器加速比
递归 fib(26)267 ms2,530 ms9.5x
循环累加 200 万次514 ms6,203 ms12.1x

性能短板（如实说明）

  • 并非「接近 C 的性能」：等价 rustc -O 跑 2000 万次循环约 274 ms，AOT 版约 2,962 ms——慢约 10 倍。
  • 原因是生成的 C 对每次整数运算都做装箱（hn_int()）、 深拷贝（hn_copy()）与释放，例如 hn_truthy(hn_le_v(hn_copy(v0), hn_int(1)))，gcc 无法优化成寄存器运算。
  • 这是「列表/字典值语义 + 统一装箱 HnValue」的必然代价， 调高优化级别（-O3）无效。纯计算密集的热循环建议用 hone prof 定位后改用 C 扩展（hone build --dll）。

两个 C 后端的共用与分界

--exe -c（AOT）与 --dll 都在做「Hone AST → C」， 但值模型完全不同：AOT 统一装箱 struct HnValue（union + 引用计数）， --dll 用 int64_t / double / const char* 直连 C ABI。 因此代码生成主体不共用——强制合并需要让 --dll 也装箱， 会破坏所有既有调用方。共用的只是与值模型无关的机械部分， 收敛在 src/cgen_util.rs（字符串字面量转义等）。

这一收敛修掉过一个静默 bug：--dll 原先用十六进制 \x01 转义控制字符，而 C 的 \x 转义是贪婪的—— "\x01a" 会被解析成单字节 \x1a， 含控制字符的字符串会被截短、内容错误且不报编译错误。现统一为三位八进制。

交互式解释器 hone repl

hone repl 启动 Python 式交互环境：逐行执行、自动回显表达式结果， 跨输入共享变量与函数定义，适合快速实验、验证语法与调试算法。

#11. 错误处理

错误统一为 error[Hxxx] 格式：编译期静态检查报错，运行期可恢复错误用 try-catch 捕获。

error[H001]: type mismatch: variable `x` is locked to `int`, got `str`
  --> err.hn:3:1
3 | x = "Hone";
  | ^
help: Hone types are locked after inference; no implicit conversion is allowed
错误码含义
H001 ~ H012类型 / 语法 / 推导 / 运行期基础错误（含泛型同一类型参数处实参不一致、类方法裸名调用、类型参数重复声明等，见下）
H101 ~ H104词法语法细分：非法字符 / 未闭合字符串 / 未闭合注释 / 缺少分号
H105 / H106控制流标签与宏：H105 标签/跳转错误（未定义标签、标签重复、跨函数跳转、跳进内层语句块、向前跳过变量绑定）；H106 宏定义/展开错误（非顶层定义、重复定义、与函数/类/枚举重名、实参个数不符、语句宏用于表达式位置、宏体含 return/goto/lambda 等越界用法）
H200 ~ H204网络：请求失败 / 超时 / 连接拒绝 / DNS 失败 / 非 2xx 响应（http_get / http_post / http.request / smtp.send / ws.request）
H300 ~ H303系统 / DLL：系统调用 / 库加载失败（含 sqlite 库缺失）/ 参数校验 / 权限不足
H304 ~ H305指针：野指针（未分配/已释放/重复释放/空指针）/ 越界访问（ptr 类）
H306标准输入：读取失败 / EOF（input / read_int / read_float）
H401 ~ H404文件：不存在 / 权限不足 / 被占用锁定 / 通用 NOT_FOUND
H600throw 主动抛出的用户错误
H700assert 断言失败（测试框架）
H999尚未实现

class / 泛型 / 新模块相关报错：

  • H001 泛型：同一类型参数处实参类型不一致（如 pick[T](a: T, b: T) 两次实参需同型）
  • H002 类方法裸名调用（成员函数不进全局符号表，必须 类.方法(...)）；泛型注解引用未声明的类型参数
  • H005 类型参数重复声明（fn dup[T, T]）、类名与 struct 名重复、类成员非 fn 定义
  • H011 struct 构造 / 类方法调用的参数个数不符
  • H301 sqlite 库缺失：Linux 安装 libsqlite3，Windows 将 sqlite3.dll 放到 hone.exe 旁边

完整解释与修复建议：hone explain H201。

try-catch 恢复：捕获可恢复错误（ZError 通道）；panic 类崩溃不进入捕获。 配合 --restart / --resume 实现无人值守运行。

try {
    resp = http_get("https://example.com/data");
} catch e {
    print(e.code + ": " + e.message);  // 字段: code/message/file/line/col/context
    resp = db.get("cached");        // 降级回退
}

throw "用户错误";   // 主动抛出 → error[H600]
throw e;             // 重抛已捕获的 error 值
机制说明
--restart[=N]脚本失败自动重拉（默认最多 3 次）；--backoff=a,b,c 重启间隔递增（默认 1,3,10 秒）；--restart-on=H200,H401 只对白名单错误码重启，确定性错误直接退出
--resume恢复上次 db 检查点；db.set 自动落盘 ~/.hone/state/，崩溃不丢检查点，脚本内容变更后检查点自动失效
错误对象字段e.code / e.message / e.file / e.context（str）；e.line / e.col（int）

注意：Hone 为块作用域，try/catch 块内赋值的变量在块外不可见，跨块共享的变量需在块前声明。示例见 examples/try_catch.hn、examples/restart_demo.hn、examples/resume_demo.hn。

#12. 设计原则

  • 不自举：Hone 不用于开发自己的编译器，保持设计与实现的灵活性
  • 不背兼容包袱：旧版本永久可用，新版本自由演化
  • 无前缀声明：变量不需要 let / var / const
  • 静态强类型：类型锁定，不隐式转换，报错精准
  • 单文件分发：下载 → 解压 → 运行，零配置；多版本共存互不干扰

#13. 图形界面库（hone_lib/gui.hn）

纯 Hone 编写的跨平台 GUI：浏览器渲染 + 本地 HTTP 服务器双向交互（Windows / Linux / Termux 均有浏览器）。依赖 server.* 内置函数。

import "gui" from "./hone_lib/gui.hn";

// 界面事件处理：id 为控件 id，value 为用户输入值（约定函数名 on_event）
fn on_event(id : str, value) -> str {
    if (id == "btn_hi") {
        return json_stringify({"update": [["lbl_out", "你好，Hone GUI！"]]});
    }
    return "";
}

widgets = [
    gui_button("btn_hi", "打招呼"),
    gui_input("txt_name", "姓名", "输入名字"),
    gui_label("lbl_out", "(事件输出)"),
];
gui_run("Hone GUI 演示", widgets);  // 启动本地服务器并自动打开浏览器，Ctrl+C 退出

控件：gui_button gui_label gui_input gui_select gui_html；on_event 返回值按 JSON 协议解释：{"update": [[元素id, 新文本], ...]} 更新元素、{"alert": "消息"} 弹窗、其他文本显示在页面底部状态栏。

底层 server.* API：server.listen(port) 启动后台监听（0=自动分配，返回实际端口）、server.poll() 取出排队请求（返回 JSON 数组）、server.respond(id, body) 发送响应体（HTTP 200）。事件模型为后台线程收发 + 脚本主线程轮询响应，与解释器单线程模型兼容。示例：examples/gui_demo.hn、examples/server_demo.hn。

#14. 进程检测库（hone_lib/process.hn）

纯 Hone 编写的跨平台进程检测库（Windows / Linux / macOS）：枚举进程、按名称/PID 查找、结束进程。Hone 无内置进程 API，本库经 sys.run 调用系统命令实现（Windows 用 tasklist / taskkill，Unix 用 ps / kill -9）。查询函数返回进程对象 {pid, name, mem, cmdline}（mem 为 KB；cmdline 仅 Unix 有值）。

import "process" from "./hone_lib/process.hn";

if (proc_exists("hone.exe")) { print("hone 正在运行"); }

// 枚举 / 模糊查找（不区分大小写）
for p in proc_find("chrome") {
    print(p.pid, p.name, p.mem);
}

// 结束进程（taskkill /F /PID 或 kill -9）
proc_kill(1234);
n = proc_kill_by_name("notepad.exe");  // 返回成功数

函数：proc_os proc_list proc_find proc_exists proc_by_pid proc_count proc_pids proc_names proc_kill proc_kill_by_name。跨平台差异：Windows 进程名含 .exe、Unix 不含，用子串匹配（如 proc_find("chrome")）两边通用。完整函数表与测试脚本见 标准库 · process 进程模块。

#15. 定时任务库（hone_lib/sched.hn）

纯 Hone 编写的跨平台定时任务库（Windows / Linux / macOS）：按间隔秒、每天指定时刻、或 cron 表达式调度命令执行。 任务动作以「命令字符串 + sys.run」描述——命令字符串便于调度与序列化，与 cron 心智模型一致。

import "sched" from "./hone_lib/sched.hn";

tasks = [
    sch_every(60, "hone backup.hn", "backup"),          // 每分钟备份
    sch_at("09:30", "hone report.hn", "daily-report"),  // 每天 9:30
    sch_cron("0 2 * * 1", "hone cleanup.hn", "weekly"), // 每周一凌晨 2 点
];
sch_run(tasks, 0);   // 一直运行，Ctrl+C 退出

函数：sch_every sch_at sch_cron sch_run sch_tick sch_next_hhmm sch_cron_match sch_cron_next sch_now sch_hhmm_to_sec。cron 为 5 段表达式（分 时 日 月 周），字段支持 * / 数字 / a-b / */n / a-b/n / 逗号列表。注意列表/字典是值语义，sch_run 返回重建后的任务列表需重新赋值。完整函数表与测试脚本见 标准库 · sched 定时任务模块。

#16. 图片库（hone_lib/img.hn）

纯 Hone 编写的图片标准库：像素网格 + 绘图/滤镜/变换 + PPM P3 / SVG 读写。 图像表示为 dict {"w": 宽, "h": 高, "rows": [[[r,g,b],...],...]}， rows[y][x] 为像素（各通道 0-255，原点在左上角）。 Hone 的 str 无法承载二进制（见手册 §3.19），故采用文本格式：PPM P3 （ASCII 光栅，GIMP/ImageMagick 可查看）与 SVG（文本矢量，浏览器直接打开），不支持 PNG/BMP。 列表为值类型、索引赋值整体拷贝，绘图/滤镜适合 ≤128×128 的小图。

import "img" from "./hone_lib/img.hn";

im = img_new(64, 64, 255, 255, 255);         // 白色画布
im = img_rect(im, 8, 8, 40, 40, 255, 0, 0);  // 红色实心矩形
im = img_circle(im, 32, 32, 16, 0, 0, 255);  // 蓝色实心圆
im = img_grayscale(im);                          // 灰度滤镜
img_save_ppm(im, "out.ppm");                    // 保存 PPM P3
img_save_svg(im, "out.svg", 4);              // 保存 SVG（每像素放大 4 倍）

函数：创建/基础 img_new img_fill img_get img_set img_size； 绘图 img_rect img_line img_circle img_gradient_h/v img_checker img_noise； 滤镜 img_grayscale img_invert img_brightness img_contrast img_threshold； 变换 img_flip_h/v img_rotate90 img_crop img_scale； 读写 img_to_ppm img_save_ppm img_from_ppm img_load_ppm img_to_svg img_save_svg。 完整函数表与测试脚本见 标准库 · img 图片模块。

#17. 桌宠库（hone_lib/pet.hn）

Windows 桌面宠物标准库（仅支持 Windows，需 PowerShell 5.1+）：Hone 驱动状态机 + PowerShell 透明置顶窗渲染。内置像素猫 8 帧动画（idle 眨眼 / walk 走路镜像 / sleep 睡觉 / happy 开心 / surprise 惊讶），说话气泡、点击/拖拽交互、右键菜单（换装/静音/跟随鼠标/ 隐藏显示台词/退出）、全屏游荡、自动入睡、整点报时、CPU/内存播报（wmic）。 提供阻塞 pet_run(cfg) 与非阻塞 pet_tick(st) 两种形态。 运行示例：hone examples/pet_demo.hn（Ctrl+C 或右键菜单退出）。

import "pet" from "./hone_lib/pet.hn";

cfg = {"pal": 0, "scale": 2, "speed": 3, "sleep_after": 90};
pet_run(cfg);   // 阻塞运行（Ctrl+C 或右键菜单退出）

// 非阻塞：st = pet_init(cfg); while (st.quit == false) { st = pet_tick(st); time.sleep(st.cfg.tick); }

配置项（均可省略，用默认值）：pal 调色板（0 橘猫/1 黑猫/2 白猫）、 scale 显示缩放（默认 2 → 96×96 窗口）、speed 移动速度、 tick 帧间隔（默认 0.125 = 8fps）、sleep_after 入睡阈值秒、 wander 游荡间隔秒、bubble 随机台词间隔秒、 mute 静音、hourly 整点报时、 res_report 资源播报间隔秒、lines_file 自定义台词 JSON、 frames_dir 外部 PPM 帧目录、start_x/start_y 初始位置。 帧构建用 ptr.alloc 缓冲 + 24×24 内联整数判定 + 2× 最近邻放大到 48×48，8 帧约 2s。 完整函数表与配置说明见 标准库 · pet 桌宠模块。

#18. 原生图形界面库（hone_lib/guipro.hn）

原生窗口 + 原生控件标准库（gui.hn 的升级版，§13 为浏览器方案）：直接创建系统原生窗口与控件。 Windows 用 Win32 标准控件（user32/gdi32/commctrl，零新增依赖、不增体积），Linux 运行时优先动态加载 GTK3 （libgtk-3.so），缺失时自动回退 X11 自绘后端（libX11.so.6 动态加载，单窗口自绘）。 Rust 内置 guipro.* 原语 + Hone 层统一 API：窗口 / button / label / input / select / checkbox / radio / slider / table / tree / canvas 控件、VBox 布局、闭包事件分发、定时器、消息框、托盘图标（XEmbed）、菜单栏。 运行示例：hone examples/guipro_demo.hn（基础控件，20 秒后自动关闭演示）、 hone examples/guipro_adv_demo.hn（进阶控件：滑块/表格/树/画布/托盘/菜单）。

import "guipro" from "./hone_lib/guipro.hn";

win = guipro_window("示例", 420, 320);
b1  = guipro_button(win, "点我", 10, 10, 100, 32);
out = guipro_label(win, "尚未点击", 10, 52, 200, 24);
handlers = [];
handlers = guipro_on(handlers, win, b1, "click", fn(id, value) {
    guipro_set_text(win, out, "点击了按钮 #" + to_str(id));
});
guipro_run(handlers);   // 阻塞主循环，窗口关闭后返回

控件：guipro_button/label/input/select/checkbox/radio(win, ..., x, y, w, h) 绝对定位， guipro_vbox(win, specs, x, y, gap) 纵向布局（specs 为控件 dict 列表，返回控件 id 列表）。 Linux GTK3 后端忽略 x/y，控件自动纵向排布；X11 自绘后端按绝对定位。

进阶控件：滑块 guipro_slider(win, min, max, value, x, y, w, h)（读写 guipro_set_value/get_value，change 事件 value=当前位置）、表格 guipro_table(win, columns, rows, x, y, w, h)（读写 guipro_table_add_row/clear/count/get/get_row/set，change 事件 value=选中行索引，click 双击同）、树 guipro_tree(win, items, x, y, w, h)（读写 guipro_tree_add/clear/get， change 事件 value=选中节点 id）、画布 guipro_canvas(win, x, y, w, h)（绘制 guipro_canvas_line/rect/ellipse/text/clear/repaint，click 事件 value="[x,y]"， 颜色支持 "#RRGGBB" 字符串或整数 RGB）。

托盘与菜单：guipro_tray_add/tray_tip/tray_remove 托盘图标 （Windows 用 Shell_NotifyIcon；Linux X11 用 XEmbed 系统托盘协议，需托盘管理器），tray 事件 value=left/right/double； guipro_menu(win, items) 菜单栏（items 为嵌套列表，text 为 "-" 表示分隔线）， menu 事件 value=菜单路径（如 "文件/打开"）。

事件与定时器：guipro_on(handlers, win, id, evt, fn) 注册 click/change 回调 （fn(id, value)，value 为文本/勾选态 "1"/"0"），guipro_on_win(handlers, win, "close", fn) 窗口级事件，guipro_timer(handlers, ms, fn) 周期定时器（fn 无参）； guipro_run(handlers) 阻塞分发，窗口 close 后返回。

架构：内置 builtins::call 无解释器上下文，闭包无法从 Rust 直接回调， 故 Rust 层只推送事件队列（guipro.poll() 返回 JSON，含 click/change/close/resize/menu/tray 事件）， 闭包分发在 Hone 层主循环完成；注册表函数式传递（Hone 函数内不能改全局变量）。 完整函数表与事件协议见 标准库 · guipro 原生图形界面模块。

#19. 进阶用法（多文件 / 调试 / 性能）

19.1 多文件项目组织

单文件即可运行，项目变大后建议拆分模块。三种复用方式：

// import：.hn 源码复用（最常用）
import "math" from "./hone_lib/math.hn";   // 本地：相对当前目录
import "mod" from "https://.../mod.hn";   // 远程：缓存到 ~/.hone/cache/
import "mod" from "https://.../mod.hn" as m; // 别名：前缀替换为 m

// load：C 动态库（typed 签名块见第 9 章）
load "libm.so.6" as m {
    fn cos(x: float) -> float;
}

模块函数以「模块名_」为前缀注册（如 math_clamp、coll_sum）， as 别名可整体替换前缀。函数名全局唯一、不支持嵌套遮蔽，命名加前缀避免冲突。 推荐目录：入口 main.hn + libs/ 自有模块 + data/ 数据； 用 hone get <script.hn> 预下载脚本中所有 import 的模块， 或在项目根目录放 hone.json 清单（name / version / modules），用 hone get 批量下载全部模块（见 §15 工具链）。

19.2 函数别名（alias）

给已有函数起新名字再调用，原名保持不变：

// 基础：alias 原名 as 新名;  之后用新名调用
fn greet(name) { return "你好 " + name; }
alias greet as hi;
hi("Hone");          // 你好 Hone

// 原名支持点号路径（模块 / 类 / 内置点号函数）
alias random.int as rint;
rint(1, 100);         // [1, 100] 随机整数

// 别名可叠加（别名再起别名）、可指向内置函数
alias greet as hello;
alias hello as hey;
alias print as p;
p("OK");

别名作用域为文件级（当前 .hn 文件全局可见），在 alias 语句之后即可用新名调用。 完整示例见 examples/alias_demo.hn。

19.4 性能优化

  • 避免循环内反复 append：列表是值类型，append 返回新列表（整体拷贝）， 循环里 l = append(l, x) 是 O(n²)。累积数据时优先用 dict，或先确定规模再一次性构造
  • 字符串拼接优先 f-string：f"..." 比 + 简洁；str 不可变，大量拼接每次都会生成新串
  • 减少深拷贝：clone / copy 是递归深拷贝，只在需要独立副本时使用
  • 热点计算交给 C：瓶颈函数用 load 调用 C 库（FFI），或用 hone build --dll 导出后由 C 调用
  • 并行化：go 函数名(参数); 启动独立线程（值传递、不共享变量）， 适合可拆分的独立任务；线程间通信走文件 / HTTP / db
  • 先用 hone prof 找热点：剖析模式输出各函数总耗时排名，优先优化耗时最长的； 再配合 hone poop 评估嵌套与圈复杂度；优化前先用 hone test 固化行为
