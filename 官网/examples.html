#示例代码

按知识点分组，与语言手册章节对应。以下示例均可在 examples/ 目录中找到，用 hone run <file.hn> 运行。

#1. 基础

1.1 Hello World
第一个 Hone 程序
// 这是 Hone 的第一个示例
print("Hello Hone!");
1.2 变量与类型
自动推导、显式类型、类型锁定（期望输出：60 / 3.14 / int / 3）
x = 10;            // 自动推导为 int
y : int = 20;    // 显式类型（Rust/TS 风格）
int z = 30;      // 显式类型（C 风格）
print(x + y + z);   // 60

f = 3.14;
print(to_str(f));  // 3.14
print(type_of(x)); // int

a = 7 / 2;
print(a);           // 3（整数除法）
// x = "Hone";  // error[H001]：类型锁定

#2. 控制流

2.1 if / while
条件必须是 bool；支持 else if 链（期望输出：45 / B）
i = 0; sum = 0;
while (i < 10) {
    sum = sum + i;
    i = i + 1;
}
print(sum);  // 45

score = 85;
if (score >= 90) {
    print("A");
} else if (score >= 80) {
    print("B");   // 输出 B
} else {
    print("C");
}
2.2 for-in 循环
遍历列表（单变量）与字典（键、值双变量）
sum = 0;
for x in [1, 2, 3] {
    print(x * x);          // 1 4 9
    sum = sum + x;
}

d = {"name": "hone", "ver": 1};
for k, v in d {
    print(k + ": " + to_str(v));
}
print(sum);              // 6
2.3 C 风格 for / do-while / continue
for (init; cond; step) 三段可省略；do-while 至少执行一次；continue 跳过本次循环（期望输出：0 1 2 3 4 / 3 / 1 3 5）
// C 风格 for：init; cond; step 三段均可省略（for (;;) 无限循环，配合 break 退出）
for (i = 0; i < 5; i = i + 1) {
    print(i);            // 0 1 2 3 4
}

// do-while：先执行循环体再判断条件（至少执行一次）
j = 0;
do { j = j + 1; } while (j < 3);
print(j);              // 3

// continue：跳过本次循环剩余语句，只打印奇数
for (i = 1; i <= 5; i = i + 1) {
    if (i % 2 == 0) { continue; }
    print(i);            // 1 3 5
}
2.4 复合赋值与自增/自减
+= -= *= /= %=（str 仅支持 += 拼接）；i++ 后缀返回旧值，++i 前缀返回新值（期望输出：2 / Hone! / 1 / 1 / 3）
x = 10;
x += 5; x -= 3; x *= 2; x /= 4; x %= 4;
print(x);              // 2

s = "Hone";
s += "!";
print(s);              // Hone!

i = 0;
i++;
print(i);              // 1（后缀返回旧值）
print(i++);            // 1（i 变为 2）
print(++i);            // 3（前缀返回新值）

#3. 函数

3.1 递归函数：斐波那契
参数类型可由调用上下文推导（期望输出：55 / 42）
fn fib(n) {
    if (n <= 1) {
        return n;
    }
    return fib(n - 1) + fib(n - 2);
}

print(fib(10));  // 55

fn add(a : int, b : int) -> int {
    return a + b;
}
print(add(20, 22));  // 42
3.2 函数默认参数
省略尾部实参取默认值；默认表达式调用时求值、可引用前面参数；必选参数必须在前（期望输出：Hello, Hone! / Hi, Hone! / *-*-* / *-* / *+*+*+*）
fn greet(name, greeting = "Hello") {
    return greeting + ", " + name + "!";
}
print(greet("Hone"));           // Hello, Hone!
print(greet("Hone", "Hi"));     // Hi, Hone!

fn repeat_char(ch, times = 3, sep = "-") {
    s = "";
    for (i = 0; i < times; i = i + 1) {
        s = s + ch;
        if (i < times - 1) { s = s + sep; }
    }
    return s;
}
print(repeat_char("*"));          // *-*-*
print(repeat_char("*", 2));       // *-*
print(repeat_char("*", 4, "+"));  // *+*+*+*
3.3 匿名函数（lambda 闭包）
一等值：赋值/传参/返回；创建时按值捕获外围变量（期望输出：42 / 15 / 36 / 15）
double = fn(x) { return x * 2; };
print(double(21));     // 42

base = 10;
adder = fn(y) { return base + y; };  // 捕获 base（按值）
print(adder(5));       // 15

fn apply(f, v) { return f(v); }
print(apply(fn(n: int) { return n * n; }, 6));  // 36

fn make_adder(offset) { return fn(z) { return z + offset; }; }
add5 = make_adder(5);
print(add5(10));       // 15

新语法综合示例（continue / C 风格 for / do-while / 复合赋值 / 自增自减 / 三元表达式 / 空值合并 / lambda 闭包 / 函数默认参数 / 三引号原始字符串）见 examples/new_syntax_demo.hn。

#4. 集合与字符串

4.1 列表操作
列表字面量、len/append/contains/index_of（期望输出：3 / [1, 2, 3, 4] / true / false / 3 / -1）
l = [1, 2, 3];
print(len(l));           // 3
l = append(l, 4);        // [1, 2, 3, 4]
print(to_str(l));
print(contains(l, 3));   // true
print(contains(l, 9));   // false
print(index_of(l, 4));   // 3
print(index_of(l, 99));  // -1
4.2 字典操作
字典字面量、has_key/keys/values/len（期望输出：{name: hone, ver: 1} / true / false / [name, ver] / [hone, 1] / 2）
d = {"name": "hone", "ver": 1};
print(to_str(d));          // {name: hone, ver: 1}
print(has_key(d, "name")); // true
print(has_key(d, "nope")); // false
print(to_str(keys(d)));    // [name, ver]
print(to_str(values(d)));  // [hone, 1]
print(len(d));           // 2（键个数）
4.3 f-string 字符串插值
f"..." 内 {expr} 嵌入表达式，{{ / }} 转义字面大括号
name = "hone";
print(f"你好, {name}!");        // 你好, hone!

a = 3; b = 2;
print(f"sum = {a + b}, pi = {3.14}");  // sum = 5, pi = 3.14

l = [1, 2, 3];
d = {"a": 1};
print(f"list = {l}, dict = {d}");   // list = [1, 2, 3], dict = {a: 1}
4.4 类型判断
is_int / is_float / is_str / is_bool / is_list / is_dict / is_null
a = 42; b = 3.14; s = "hello"; l = [1, 2];
print(is_int(a));       // true
print(is_float(a));     // false
print(is_float(b));     // true
print(is_str(s));       // true
print(is_list(l));      // true
print(is_bool(true));  // true
4.5 结构体（struct）与深度拷贝
struct 声明确定的数据形态（字段名与类型固定），构造时按字段顺序传参并校验类型；clone/copy 深度拷贝防别名
struct Point { x: int, y: float };
struct Rect { w: int, h: int, label: str };

p = Point(3, 2.5);
print(p.x);            // 3
r = Rect(10, 20, "box");
print(r.label);        // box

// 深度拷贝：修改副本不影响原值
l1 = [1, 2, [3, 4]];
l2 = clone(l1);
l2 = append(l2, 99);
print(to_str(l1));     // [1, 2, [3, 4]]（不变）
4.6 模式匹配（match）
match 是表达式，返回匹配分支的值；模式支持字面量与 _ 通配符（放最后）；全部不匹配运行时报错
s = match 2 {
    1 => "one",
    2 => "two",
    _ => "other",
};
print(s);                 // two

status = 404;
msg = match status {
    200 => "OK",
    404 => "Not Found",
    500 => "Server Error",
    _ => "Unknown",
};
print(msg);               // Not Found
4.7 管道操作符（|>）
x |> f 等价 f(x)；x |> f(a, b) 等价 f(x, a, b)；可链式，解析期转换为普通调用
// 嵌套调用写法 vs 管道写法
print(to_str(len("hello")));      // "5"
print("hello" |> len |> to_str);  // "5"

print([1, 2, 3] |> len);          // 3
print(3 |> max(7) |> to_str);     // "7"

5.13 指针类（ptr.*）
分配 / 释放 / 读写内存；分配表跟踪防野指针（未分配/已释放/重复释放报 H304，越界报 H305）
p = ptr.alloc(32);                  // 分配 32 字节
ptr.write_int(p, 0, 12345);       // 偏移 0 写 8 字节整数
print(ptr.read_int(p, 0));        // 12345

5.14 插件系统（plugin.*）
运行期加载动态库并注册，之后可用别名直接调用（走 C ABI 通道）；plugin.list 列出已注册插件
ok = plugin.load("path/to/lib.dll", "pl");  // 加载并注册别名 pl
print(plugin.has("pl"));                  // true
print(pl.lib_add(3, 4));                  // 7（调用库函数）
print(to_str(plugin.list()));              // [{name: pl, path: ...}]
plugin.unload("pl");                       // 注销
5.15 标准输入与默认值（EOF 降级）
input / read_int / read_float 从 stdin 读取；EOF 抛 H306、数字格式错抛 H006/H007，用 try-catch 包装实现默认值（期望输出：管道喂入 "Bob\n25\n" → 你好, Bob! / 明年你 26 岁）
fn ask(prompt, fallback) {
    try {
        return input(prompt);
    } catch e {
        return fallback;   // H306 EOF → 默认值
    }
}
name = ask("姓名: ", "匿名");
print("你好, " + name + "!");

fn ask_int(prompt, fallback) {
    try {
        return read_int(prompt);
    } catch e {
        return fallback;   // H006 格式错 / H306 EOF → 默认值
    }
}
age = ask_int("年龄: ", 0);
print("明年你 " + to_str(age + 1) + " 岁");

// 完整示例：examples/input_demo.hn；EOF 也可用 ?? 空值合并？——不行，
// input 在 EOF 抛错误而非返回 null，?? 只对 null 值生效，必须 try-catch

#6. 并发

6.1 go 多线程
每次 go 启动独立线程，不共享变量；子线程崩溃不影响主线程
fn task(id) {
    i = 0;
    while (i < 100000) { i = i + 1; }
    print("任务 " + to_str(id) + " 完成");
}

go task(1);
go task(2);
go task(3);
print("主线程继续执行");

#7. 导入与外部集成

7.1 import as 别名导入
以别名导入远程模块（URL 支持 http/https），函数名前缀自动替换
import "math_mod" from "http://example.com/math_mod.hn" as m;
print(m_add(20, 22));  // 42
7.2 DLL 打包
@export 标记导出，hone build --dll 生成 C ABI 动态库（可在 C / Python / Rust 中调用）
// math.hn
fn add(a : int, b : int) -> int { return a + b; }
fn avg(a : float, b : float) -> float { return (a + b) / 2.0; }
fn greet(name : str) -> str { return "你好, " + name + "!"; }

@export add;
@export avg;
@export greet;

生成的 C ABI：int64_t add(int64_t, int64_t)、double avg(double, double)、const char* greet(const char*)。

7.3 typed FFI：load 签名块精确调用 C 函数（v0.4+）
签名块显式声明参数/返回类型（int/float/bool/str/ptr/void），按声明精确转换；调用在检查阶段校验参数（H001/H005）
load "tests/hone_lib/target/release/hone_lib.dll" as m {
    // 仅描述 C 函数 ABI，不定义 Hone 函数体；返回类型允许 void
    fn lib_add_f(a: float, b: float) -> float;
    fn lib_strlen(s: str) -> int;
    fn lib_not(b: bool) -> bool;
    fn lib_echo_ptr(p: ptr) -> ptr;
    fn lib_bump() -> void;
}

print(m.lib_add_f(1.5, 2.25));  // 3.75（float 通道）
print(m.lib_strlen("hello"));   // 5（str → int）
print(m.lib_not(true));          // false（bool）
p = m.lib_echo_ptr(0);            // ptr 句柄往返，0 为 NULL
print(p == 0);                     // true（ptr 支持 == 比较）

// 静态校验：参数类型/个数错误在检查阶段直接报错，而非运行时
// m.lib_strlen(123);   // H001: 参数期望 str，给了 int
// m.lib_strlen();      // H005: 参数个数不匹配

未声明签名的库函数保持旧 int64 通道调用（向后兼容）。完整示例见 examples/ffi_demo.hn。

7.4 头文件自动绑定：load ... from "header.h"（v0.4+）
从 C 头文件提取函数原型自动生成 FFI 签名；hone bind 可离线生成签名块粘贴进脚本
// 从头文件自动绑定：类型全部来自 hone_lib.h（等价于手写签名块）
load "tests/hone_lib/target/release/hone_lib.dll" as m from "tests/hone_lib/hone_lib.h";

print(m.lib_add(1, 2));         // 3（typedef 展开 ka_int64 → int）
print(m.lib_add_f(1.5, 2.25));  // 3.75（double → float）
print(m.lib_strlen("hello"));   // 5（const char* → str）

// 离线生成签名块（输出到 stdout，unsupported 原型列为注释）
// hone bind tests/hone_lib/hone_lib.h

// 不支持的原型（回调/变参/数组/结构体按值）调用时直接报错，而非 ABI 崩溃
// print(m.lib_cb(0));   // H006: cannot be called: callback parameter
// print(m.lib_var("x")); // H006: cannot be called: variadic (`...`)

解析能力为纯 Rust 受限子集（无 libclang 依赖）：跳过注释/预处理/struct 定义/extern "C" 块，typedef 简单展开， 属性宏跳过；类型映射 int/size_t→int、float/double→float、bool→bool、char*→str、其余指针→ptr、void→void。 完整示例见 examples/ffi_header.hn。

7.5 函数别名：alias 原名 as 新名（v0.6.1+）
给已有函数起新名字再调用；原名支持点号路径（模块/类/内置函数），别名可叠加、可指向内置函数
fn greet(name : str) -> str { return "你好 " + name; }

alias greet as hi;           // 用户函数别名
print(hi("Hone"));               // 你好 Hone

alias random.int as rint;     // 点号路径原名（内置点号函数）
print(rint(1, 100));            // [1, 100] 随机整数

alias greet as hello;          // 别名叠加：hello → greet
alias hello as hey;           // hey → hello → greet
print(hey("World"));             // 你好 World

alias abs as mabs;             // 内置函数别名
print(mabs(-5));                 // 5

别名作用域为文件级，alias 语句之后即可用新名调用。完整示例见 examples/alias_demo.hn。

#9. 错误处理

9.1 try-catch 错误捕获与恢复
捕获可恢复错误并降级回退；throw 主动抛出；错误对象字段 e.code / e.message / e.line
r = 0;
try {
    r = 10 / 0;              // error[H009] 除零
} catch e {
    print("caught " + e.code + ": " + e.message);
    r = -1;                  // 回退值
}
print(r);                   // -1

throw "自定义错误";        // error[H600]
9.2 自动重启与检查点恢复
--restart 失败自动重拉（名刀机制）；--resume 从 db 检查点继续（崩溃不丢进度）
// 失败自动重启(默认最多3次,间隔1s/3s/10s递增;--restart-on 错误码白名单)
hone run --restart=2 --backoff=1,1 --restart-on=H401 app.hn

// 检查点恢复(db.set 自动落盘 ~/.hone/state/,脚本变更后检查点自动失效)
hone run --resume app.hn

// app.hn 中写入检查点
db.set("stage", "2");

#10. 图形界面与本地服务器

10.1 图形界面库（浏览器渲染 + 双向交互）

import "gui" from "./hone_lib/gui.hn";

// 界面事件处理：id 为控件 id，value 为用户输入值（约定函数名 on_event）
fn on_event(id : str, value) -> str {
    if (id == "btn_hi") {
        return json_stringify({"update": [["lbl_out", "你好，Hone GUI！"]]});
    }
    return "";
}

widgets = [
    gui_button("btn_hi", "打招呼"),
    gui_input("txt_name", "姓名", "输入名字"),
    gui_label("lbl_out", "(事件输出)"),
];
gui_run("Hone GUI 演示", widgets);  // 浏览器自动打开，Ctrl+C 退出
10.2 本地 HTTP 服务器（server.* 内置函数）
server.listen / server.poll / server.respond：纯 std::net 跨平台，后台线程收发、脚本主线程轮询响应；respond 可指定状态码（如 404/500）
port = server.listen(0);  // 0 = 自动分配端口，返回实际端口
print("http://127.0.0.1:" + to_str(port) + "/");

while (true) {
    reqs = server.poll();   // 取排队请求: JSON 数组 [{id,method,path,body}, ...]
    if (reqs != "[]") {
        list = json_parse(reqs);
        for req in list {
            // gui_dict_get(req, "id") 取字段（Hone 无 d["k"] 索引语法）
            server.respond(1, "ok", 200);     // 指定 HTTP 200
            server.respond(2, "not found", 404); // 404（默认 200，范围 100..=599）
        }
    }
    time.sleep(0.05);
}

10.4 原生图形界面库（Windows: Win32 / Linux: GTK3）
hone_lib/guipro.hn：原生窗口 + 原生控件（gui.hn 的升级版，不依赖浏览器）；Windows 用 Win32 标准控件，Linux 动态加载 GTK3；闭包事件分发 + 定时器 + 消息框

win = guipro_window("guipro 演示", 460, 420);

// VBox 纵向布局：返回控件 id 列表（与 specs 顺序对应）
specs = [];

specs = append(specs, {"type": "select", "options": ["北京", "上海", "广州"], "w": 140, "h": 28});
specs = append(specs, {"type": "button", "text": "提交", "w": 120, "h": 32});
ids = guipro_vbox(win, specs, 24, 20, 6);

handlers = [];

    return "";
});
handlers = guipro_on_win(handlers, win, "close", fn(id, value) {
    print("窗口已关闭");
    return "";
});
guipro_run(handlers);   // 阻塞主循环，窗口关闭后返回
