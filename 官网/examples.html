#示例代码

以下示例均可在 examples/ 目录中找到，用 zap run <file.zp> 运行。

1. Hello World
第一个 Zap 程序
// 这是 Zap 的第一个示例
print("Hello Zap!");
2. 变量与类型
自动推导、显式类型、类型锁定（期望输出：60 / 3.14 / int / 3）
x = 10;            // 自动推导为 int
y : int = 20;    // 显式类型（Rust/TS 风格）
int z = 30;      // 显式类型（C 风格）
print(x + y + z);   // 60

f = 3.14;
print(to_str(f));  // 3.14
print(type_of(x)); // int

a = 7 / 2;
print(a);           // 3（整数除法）
// x = "Zap";  // error[Z001]：类型锁定
3. 递归函数：斐波那契
参数类型可由调用上下文推导（期望输出：55）
fn fib(n) {
    if (n <= 1) {
        return n;
    }
    return fib(n - 1) + fib(n - 2);
}

print(fib(10));  // 55

fn add(a : int, b : int) -> int {
    return a + b;
}
print(add(20, 22));  // 42
4. 控制流：if / while
条件必须是 bool；支持 else if 链（期望输出：45 / B）
i = 0; sum = 0;
while (i < 10) {
    sum = sum + i;
    i = i + 1;
}
print(sum);  // 45

score = 85;
if (score >= 90) {
    print("A");
} else if (score >= 80) {
    print("B");   // 输出 B
} else {
    print("C");
}
5. go 多线程
每次 go 启动独立线程，不共享变量；子线程崩溃不影响主线程
fn task(id) {
    i = 0;
    while (i < 100000) { i = i + 1; }
    print("任务 " + to_str(id) + " 完成");
}

go task(1);
go task(2);
go task(3);
print("主线程继续执行");
6. 时间与随机
时间格式化（UTC）与闭区间随机数
t = time.now();
print(time.format(t, "YYYY-MM-DD HH:mm:SS"));

r = random.int(1, 100);   // 闭区间 [1, 100]
print("随机数: " + to_str(r));
print("随机小数: " + to_str(random.float()));

time.sleep(0.5);  // 暂停 0.5 秒
7. JSON 解析与序列化
标量值自动映射为 int / float / bool / str
n = json_parse("42");
print(type_of(n));          // int

f = json_parse("3.14");
print(f);                     // 3.14

b = json_parse("true");
print(b);                     // true

print(json_stringify(100));  // 100
8. DLL 打包
@export 标记导出，zap build --dll 生成 C ABI 动态库（可在 C / Python / Rust 中调用）
// math.zp
fn add(a : int, b : int) -> int { return a + b; }
fn avg(a : float, b : float) -> float { return (a + b) / 2.0; }
fn greet(name : str) -> str { return "你好, " + name + "!"; }

@export add;
@export avg;
@export greet;

生成的 C ABI：int64_t add(int64_t, int64_t)、double avg(double, double)、const char* greet(const char*)。

9. 断点调试
zap debug 模式下 breakpoint 打印所有变量的类型与值快照
x = 10;
name = "Zap";
flag = true;
breakpoint;   // [Zap Debug] 断点触发 -> debug.zp:4 + 变量快照
print(x);
