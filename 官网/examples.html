#示例代码

按知识点分组，与语言手册章节对应。以下示例均可在 examples/ 目录中找到，用 zap run <file.zp> 运行。

#1. 基础

1.1 Hello World
第一个 Zap 程序
// 这是 Zap 的第一个示例
print("Hello Zap!");
1.2 变量与类型
自动推导、显式类型、类型锁定（期望输出：60 / 3.14 / int / 3）
x = 10;            // 自动推导为 int
y : int = 20;    // 显式类型（Rust/TS 风格）
int z = 30;      // 显式类型（C 风格）
print(x + y + z);   // 60

f = 3.14;
print(to_str(f));  // 3.14
print(type_of(x)); // int

a = 7 / 2;
print(a);           // 3（整数除法）
// x = "Zap";  // error[Z001]：类型锁定

#2. 控制流

2.1 if / while
条件必须是 bool；支持 else if 链（期望输出：45 / B）
i = 0; sum = 0;
while (i < 10) {
    sum = sum + i;
    i = i + 1;
}
print(sum);  // 45

score = 85;
if (score >= 90) {
    print("A");
} else if (score >= 80) {
    print("B");   // 输出 B
} else {
    print("C");
}
2.2 for-in 循环
遍历列表（单变量）与字典（键、值双变量）
sum = 0;
for x in [1, 2, 3] {
    print(x * x);          // 1 4 9
    sum = sum + x;
}

d = {"name": "zap", "ver": 1};
for k, v in d {
    print(k + ": " + to_str(v));
}
print(sum);              // 6

#3. 函数

3.1 递归函数：斐波那契
参数类型可由调用上下文推导（期望输出：55 / 42）
fn fib(n) {
    if (n <= 1) {
        return n;
    }
    return fib(n - 1) + fib(n - 2);
}

print(fib(10));  // 55

fn add(a : int, b : int) -> int {
    return a + b;
}
print(add(20, 22));  // 42

#4. 集合与字符串

4.1 列表操作
列表字面量、len/append/contains/index_of（期望输出：3 / [1, 2, 3, 4] / true / false / 3 / -1）
l = [1, 2, 3];
print(len(l));           // 3
l = append(l, 4);        // [1, 2, 3, 4]
print(to_str(l));
print(contains(l, 3));   // true
print(contains(l, 9));   // false
print(index_of(l, 4));   // 3
print(index_of(l, 99));  // -1
4.2 字典操作
字典字面量、has_key/keys/values/len（期望输出：{name: zap, ver: 1} / true / false / [name, ver] / [zap, 1] / 2）
d = {"name": "zap", "ver": 1};
print(to_str(d));          // {name: zap, ver: 1}
print(has_key(d, "name")); // true
print(has_key(d, "nope")); // false
print(to_str(keys(d)));    // [name, ver]
print(to_str(values(d)));  // [zap, 1]
print(len(d));           // 2（键个数）
4.3 f-string 字符串插值
f"..." 内 {expr} 嵌入表达式，{{ / }} 转义字面大括号
name = "zap";
print(f"你好, {name}!");        // 你好, zap!

a = 3; b = 2;
print(f"sum = {a + b}, pi = {3.14}");  // sum = 5, pi = 3.14

l = [1, 2, 3];
d = {"a": 1};
print(f"list = {l}, dict = {d}");   // list = [1, 2, 3], dict = {a: 1}
4.4 类型判断
is_int / is_float / is_str / is_bool / is_list / is_dict / is_null
a = 42; b = 3.14; s = "hello"; l = [1, 2];
print(is_int(a));       // true
print(is_float(a));     // false
print(is_float(b));     // true
print(is_str(s));       // true
print(is_list(l));      // true
print(is_bool(true));  // true

#5. 内置函数

5.1 时间与随机
时间格式化（UTC）与闭区间随机数
t = time.now();
print(time.format(t, "YYYY-MM-DD HH:mm:SS"));

r = random.int(1, 100);   // 闭区间 [1, 100]
print("随机数: " + to_str(r));
print("随机小数: " + to_str(random.float()));

time.sleep(0.5);  // 暂停 0.5 秒
5.2 时间戳解析
time.parse 解析 YYYY-MM-DD[THH:MM:SS] → Unix 秒（UTC），配合 time.format 回显
t = time.parse("2025-01-01T00:00:00Z");
print(t);                                  // 1735689600
print(time.format(t, "YYYY-MM-DD HH:mm:SS"));  // 2025-01-01 00:00:00

t2 = time.parse("2025-01-01 08:30:00");   // 空格分隔也可
print(time.format(t2, "YYYY-MM-DD HH:mm:SS"));  // 2025-01-01 08:30:00
5.3 JSON 解析与序列化
标量值自动映射为 int / float / bool / str
n = json_parse("42");
print(type_of(n));          // int

f = json_parse("3.14");
print(f);                     // 3.14

b = json_parse("true");
print(b);                     // true

print(json_stringify(100));  // 100
5.4 正则匹配与替换
regex.match / regex.replace 使用标准正则表达式
ok = regex.match("^\\d+$", "12345");
print(ok);  // true

result = regex.replace("world", "hello world", "Zap");
print(result);  // hello Zap
5.5 MD5 / SHA256 哈希
crypto.md5 / crypto.sha256 计算十六进制哈希
print(crypto.md5("hello"));
print(crypto.sha256("hello"));
5.6 UUID v4 生成
uuid.new 每次生成不同的 UUID v4（形如 xxxxxxxx-xxxx-4xxx-xxxx-xxxxxxxxxxxx）
a = uuid.new();
b = uuid.new();
print(a);               // 每次运行不同
print(b);
print(a != b);          // true
5.7 路径操作
path.join / path.dirname / path.basename 跨平台路径操作
print(path.join("data", "users", "1.json"));
print(path.dirname("/home/user/file.txt"));
print(path.basename("/home/user/file.txt"));
5.8 环境变量与键值存储
env.get/set 读写环境变量，db.set/get 内存键值存储
path = env.get("PATH");
env.set("MY_VAR", "hello");
db.set("token", "abc123");
print(db.get("token"));
5.9 命令行参数解析
args.get / args.has 解析 --port 8080 或 -v 标志
port = args.get("port");
if (args.has("v")) {
    print("详细模式已开启");
}
5.10 HTTP 网络请求（http / https）
http_get / http_post 支持 http:// 与 https://（TLS 为纯 Rust 实现、内置 Mozilla 根证书）
html = http_get("https://example.com/");
print(len(html));              // 页面字节数
resp = http_post("http://httpbin.org/post", "hello");
print(str_contains(resp, "hello"));  // true
5.11 彩色日志
log.info / log.warn / log.error / log.debug 彩色输出到 stderr
log.info("程序启动完成");
log.warn("磁盘空间不足");
log.error("连接失败");
log.debug("x = 42");

#6. 并发

6.1 go 多线程
每次 go 启动独立线程，不共享变量；子线程崩溃不影响主线程
fn task(id) {
    i = 0;
    while (i < 100000) { i = i + 1; }
    print("任务 " + to_str(id) + " 完成");
}

go task(1);
go task(2);
go task(3);
print("主线程继续执行");

#7. 导入与外部集成

7.1 import as 别名导入
以别名导入远程模块（URL 支持 http/https），函数名前缀自动替换
import "math_mod" from "http://example.com/math_mod.zp" as m;
print(m_add(20, 22));  // 42
7.2 DLL 打包
@export 标记导出，zap build --dll 生成 C ABI 动态库（可在 C / Python / Rust 中调用）
// math.zp
fn add(a : int, b : int) -> int { return a + b; }
fn avg(a : float, b : float) -> float { return (a + b) / 2.0; }
fn greet(name : str) -> str { return "你好, " + name + "!"; }

@export add;
@export avg;
@export greet;

生成的 C ABI：int64_t add(int64_t, int64_t)、double avg(double, double)、const char* greet(const char*)。

#8. 工具链

8.1 build --exe 打包独立可执行文件
解释器+脚本打包为单文件，双击即用；--version 查看内嵌版本信息
// 打包(默认输出 app.exe,体积≈解释器+脚本)
zap build --exe app.zp [-o my_app.exe]

// 运行:自动释放运行时到 .zap_cache,执行完毕后自动清理
./my_app.exe

// 版本信息(内嵌 Zap 版本/构建时间/脚本名/校验和)
./my_app.exe --version
// > my_app.exe (Zap v0.2.0) built at 2026-08-07 09:24:30
// > script: app.zp

// 保留缓存以加速下次启动
./my_app.exe --keep-cache
8.2 断点调试
zap debug 模式下 breakpoint 打印所有变量的类型与值快照
x = 10;
name = "Zap";
flag = true;
breakpoint;   // [Zap Debug] 断点触发 -> debug.zp:4 + 变量快照
 print(x);
8.3 临时函数与调试输出
tmp fn 编译自动忽略，debug_print 仅 debug 模式生效
tmp fn helper() {
    print("只在开发时有用");
}

x = 42;
debug_print("当前 x = " + to_str(x));

#9. 错误处理

9.1 try-catch 错误捕获与恢复
捕获可恢复错误并降级回退；throw 主动抛出；错误对象字段 e.code / e.message / e.line
r = 0;
try {
    r = 10 / 0;              // error[Z009] 除零
} catch e {
    print("caught " + e.code + ": " + e.message);
    r = -1;                  // 回退值
}
print(r);                   // -1

throw "自定义错误";        // error[Z600]
9.2 自动重启与检查点恢复
--restart 失败自动重拉（名刀机制）；--resume 从 db 检查点继续（崩溃不丢进度）
// 失败自动重启(默认最多3次,间隔1s/3s/10s递增;--restart-on 错误码白名单)
zap run --restart=2 --backoff=1,1 --restart-on=Z401 app.zp

// 检查点恢复(db.set 自动落盘 ~/.zap/state/,脚本变更后检查点自动失效)
zap run --resume app.zp

// app.zp 中写入检查点
db.set("stage", "2");
